1//! Support for the 8259 Programmable Interrupt Controller, which handles
2//! basic I/O interrupts.  In multicore mode, we would apparently need to
3//! replace this with an APIC interface.
4//!
5//! The basic idea here is that we have two PIC chips, PIC1 and PIC2, and
6//! that PIC2 is slaved to interrupt 2 on PIC 1.  You can find the whole
7//! story at http://wiki.osdev.org/PIC (as usual).  Basically, our
8//! immensely sophisticated modern chipset is engaging in early-80s
9//! cosplay, and our goal is to do the bare minimum required to get
10//! reasonable interrupts.
11//!
12//! The most important thing we need to do here is set the base "offset"
13//! for each of our two PICs, because by default, PIC1 has an offset of
14//! 0x8, which means that the I/O interrupts from PIC1 will overlap
15//! processor interrupts for things like "General Protection Fault".  Since
16//! interrupts 0x00 through 0x1F are reserved by the processor, we move the
17//! PIC1 interrupts to 0x20-0x27 and the PIC2 interrupts to 0x28-0x2F.  If
18//! we wanted to write a DOS emulator, we'd presumably need to choose
19//! different base interrupts, because DOS used interrupt 0x21 for system
20//! calls.
21
22#![no_std]
23
24use x86_64::instructions::port::Port;
25
26/// Command sent to begin PIC initialization.
27const CMD_INIT: u8 = 0x11;
28
29/// Command sent to acknowledge an interrupt.
30const CMD_END_OF_INTERRUPT: u8 = 0x20;
31
32// The mode in which we want to run our PICs.
33const MODE_8086: u8 = 0x01;
34
35/// An individual PIC chip.  This is not exported, because we always access
36/// it through `Pics` below.
37struct Pic {
38    /// The base offset to which our interrupts are mapped.
39    offset: u8,
40
41    /// The processor I/O port on which we send commands.
42    command: Port<u8>,
43
44    /// The processor I/O port on which we send and receive data.
45    data: Port<u8>,
46}
47
48impl Pic {
49    /// Are we in charge of handling the specified interrupt?
50    /// (Each PIC handles 8 interrupts.)
51    fn handles_interrupt(&self, interrupt_id: u8) -> bool {
52        self.offset <= interrupt_id && interrupt_id < self.offset + 8
53    }
54
55    /// Notify us that an interrupt has been handled and that we're ready
56    /// for more.
57    unsafe fn end_of_interrupt(&mut self) {
58        self.command.write(CMD_END_OF_INTERRUPT);
59    }
60
61    /// Reads the interrupt mask of this PIC.
62    unsafe fn read_mask(&mut self) -> u8 {
63        self.data.read()
64    }
65
66    /// Writes the interrupt mask of this PIC.
67    unsafe fn write_mask(&mut self, mask: u8) {
68        self.data.write(mask)
69    }
70}
71
72/// A pair of chained PIC controllers.  This is the standard setup on x86.
73pub struct ChainedPics {
74    pics: [Pic; 2],
75}
76
77impl ChainedPics {
78    /// Create a new interface for the standard PIC1 and PIC2 controllers,
79    /// specifying the desired interrupt offsets.
80    pub const unsafe fn new(offset1: u8, offset2: u8) -> ChainedPics {
81        ChainedPics {
82            pics: [
83                Pic {
84                    offset: offset1,
85                    command: Port::new(0x20),
86                    data: Port::new(0x21),
87                },
88                Pic {
89                    offset: offset2,
90                    command: Port::new(0xA0),
91                    data: Port::new(0xA1),
92                },
93            ],
94        }
95    }
96
97    /// Create a new `ChainedPics` interface that will map the PIC controllers contiguously starting at the given interrupt offset.
98    ///
99    /// This is a convenience function that maps the PIC1 and PIC2 controllers to a
100    /// contiguous set of interrupts. This function is equivalent to
101    /// `Self::new(primary_offset, primary_offset + 8)`.
102    pub const unsafe fn new_contiguous(primary_offset: u8) -> ChainedPics {
103        Self::new(primary_offset, primary_offset + 8)
104    }
105
106    /// Initialize both our PICs.  We initialize them together, at the same
107    /// time, because it's traditional to do so, and because I/O operations
108    /// might not be instantaneous on older processors.
109    pub unsafe fn initialize(&mut self) {
110        // We need to add a delay between writes to our PICs, especially on
111        // older motherboards.  But we don't necessarily have any kind of
112        // timers yet, because most of them require interrupts.  Various
113        // older versions of Linux and other PC operating systems have
114        // worked around this by writing garbage data to port 0x80, which
115        // allegedly takes long enough to make everything work on most
116        // hardware.  Here, `wait` is a closure.
117        let mut wait_port: Port<u8> = Port::new(0x80);
118        let mut wait = || wait_port.write(0);
119
120        // Save our original interrupt masks, because I'm too lazy to
121        // figure out reasonable values.  We'll restore these when we're
122        // done.
123        let saved_masks = self.read_masks();
124
125        // Tell each PIC that we're going to send it a three-byte
126        // initialization sequence on its data port.
127        self.pics[0].command.write(CMD_INIT);
128        wait();
129        self.pics[1].command.write(CMD_INIT);
130        wait();
131
132        // Byte 1: Set up our base offsets.
133        self.pics[0].data.write(self.pics[0].offset);
134        wait();
135        self.pics[1].data.write(self.pics[1].offset);
136        wait();
137
138        // Byte 2: Configure chaining between PIC1 and PIC2.
139        self.pics[0].data.write(4);
140        wait();
141        self.pics[1].data.write(2);
142        wait();
143
144        // Byte 3: Set our mode.
145        self.pics[0].data.write(MODE_8086);
146        wait();
147        self.pics[1].data.write(MODE_8086);
148        wait();
149
150        // Restore our saved masks.
151        self.write_masks(saved_masks[0], saved_masks[1])
152    }
153
154    /// Reads the interrupt masks of both PICs.
155    pub unsafe fn read_masks(&mut self) -> [u8; 2] {
156        [self.pics[0].read_mask(), self.pics[1].read_mask()]
157    }
158
159    /// Writes the interrupt masks of both PICs.
160    pub unsafe fn write_masks(&mut self, mask1: u8, mask2: u8) {
161        self.pics[0].write_mask(mask1);
162        self.pics[1].write_mask(mask2);
163    }
164
165    /// Disables both PICs by masking all interrupts.
166    pub unsafe fn disable(&mut self) {
167        self.write_masks(u8::MAX, u8::MAX)
168    }
169
170    /// Do we handle this interrupt?
171    pub fn handles_interrupt(&self, interrupt_id: u8) -> bool {
172        self.pics.iter().any(|p| p.handles_interrupt(interrupt_id))
173    }
174
175    /// Figure out which (if any) PICs in our chain need to know about this
176    /// interrupt.  This is tricky, because all interrupts from `pics[1]`
177    /// get chained through `pics[0]`.
178    pub unsafe fn notify_end_of_interrupt(&mut self, interrupt_id: u8) {
179        if self.handles_interrupt(interrupt_id) {
180            if self.pics[1].handles_interrupt(interrupt_id) {
181                self.pics[1].end_of_interrupt();
182            }
183            self.pics[0].end_of_interrupt();
184        }
185    }
186}
