24/// Defines the entry point function.
25///
26/// The function must have the signature `fn(&'static BootInfo) -> !`.
27///
28/// This macro just creates a function named `_start`, which the linker will use as the entry
29/// point. The advantage of using this macro instead of providing an own `_start` function is
30/// that the macro ensures that the function and argument types are correct.
31#[macro_export]
32macro_rules! entry_point {
33    ($path:path) => {
34        #[export_name = "_start"]
35        pub extern "C" fn __impl_start(boot_info: &'static $crate::bootinfo::BootInfo) -> ! {
36            // validate the signature of the program entry point
37            let f: fn(&'static $crate::bootinfo::BootInfo) -> ! = $path;
38
39            f(boot_info)
40        }
41    };
42}
