9// Базовое смещение для прерываний аппаратной части.
10// Векторы 0..31 заняты процессором под исключения. Мапим PIC на 32..47.
11pub const PIC_1_OFFSET: u8 = 32;
12pub const PIC_2_OFFSET: u8 = PIC_1_OFFSET + 8;
13
14pub static PICS: spin::Mutex<ChainedPics> =
15    spin::Mutex::new(unsafe { ChainedPics::new(PIC_1_OFFSET, PIC_2_OFFSET) });

17// Номера векторов прерываний для наших девайсов
18#[derive(Debug, Clone, Copy)]
19#[repr(u8)]
20pub enum InterruptIndex {
21    Timer = PIC_1_OFFSET, // 32
22    Keyboard,             // 33
23}
24
25impl InterruptIndex {
26    fn as_u8(self) -> u8 {
27        self as u8
28    }
29    fn as_usize(self) -> usize {
30        self as usize
31    }
32}

34static IDT: Lazy<InterruptDescriptorTable> = Lazy::new(|| {
35    let mut idt = InterruptDescriptorTable::new();
36
37    // Исключения процессора
38    idt.breakpoint.set_handler_fn(breakpoint_handler);
39    idt.double_fault.set_handler_fn(double_fault_handler);
40
41    // Аппаратные прерывания
42    idt[InterruptIndex::Timer.as_usize()].set_handler_fn(timer_handler);
43    idt[InterruptIndex::Keyboard.as_usize()].set_handler_fn(keyboard_handler);
44
45    idt
46});
47
48pub fn init_idt() {
49    IDT.load();
50}

69// Прерывание системного таймера (тикает постоянно)
70extern "x86-interrupt" fn timer_handler(_stack_frame: InterruptStackFrame) {
71    unsafe {
72        TIMER_TICKS += 1;

80// --- БУФЕР ДЛЯ НАШЕГО TTY ---
81const CMD_BUFFER_SIZE: usize = 80;
82// Явно пишем 0u8, чтобы Rust сразу понял, что это массив байт
83static mut CMD_BUFFER: [u8; CMD_BUFFER_SIZE] = [0u8; CMD_BUFFER_SIZE];
84static mut CMD_LEN: usize = 0;
85static mut CMD_READY: bool = false;
86
87static mut SHIFT_PRESSED: bool = false;
88
89/// Функция для извлечения готовой команды (исправили опечатку в названии!)
90pub fn get_command() -> Option<[u8; CMD_BUFFER_SIZE]> {
91    x86_64::instructions::interrupts::without_interrupts(|| unsafe {
92        if CMD_READY {
93            let res = CMD_BUFFER;
94            // И здесь заменяем на чистый инициализатор без 'as'
95            CMD_BUFFER = [0u8; CMD_BUFFER_SIZE];
96            CMD_LEN = 0;
97            CMD_READY = false;
98            Some(res)
99        } else {
100            None
101        }
102    })
103}
104
105pub fn get_uptime_seconds() -> u64 {
106    unsafe { TIMER_TICKS / 18 }
107}

111// Прерывание клавиатуры (вызывается при нажатии и отпускании клавиш)
112extern "x86-interrupt" fn keyboard_handler(_stack_frame: InterruptStackFrame) {
113    use x86_64::instructions::port::Port;
114
115    let mut port = Port::new(0x60);
116    let scancode: u8 = unsafe { port.read() };
117
118    unsafe {
119        match scancode {
120            0x4b => {
121                // Стрелка влево нажата
122            }
123            0x4d => {
124                // Стрелка вправо нажата
125            }
126            // Нажатие Левого или Правого Shift
127            0x2a | 0x36 => {
128                SHIFT_PRESSED = true;
129            }
130            // Отпускание Левого или Правого Shift (скан-код + 0x80)
131            0xaa | 0xb6 => {
132                SHIFT_PRESSED = false;
133            }
134            // Обрабатываем остальные нажатия клавиш (коды < 0x80)
135            0x00..=0x7f => {
136                if !CMD_READY {
137                    match scancode {
138                        // Нажатие Enter
139                        0x1c => {
140                            print!("\n");
141                            CMD_READY = true;
142                        }
143                        // Нажатие Backspace
144                        0x0e => {
145                            if CMD_LEN > 0 {
146                                CMD_LEN -= 1;
147                                CMD_BUFFER[CMD_LEN] = 0;
148                                crate::vga_buffer::backspace();
149                            }
150                        }
151                        // Все остальные символы
152                        _ => {
153                            let key = match scancode {
154                                // Цифровая строка и символы
155                                0x02 => Some(if SHIFT_PRESSED { b'!' } else { b'1' }),
156                                0x03 => Some(if SHIFT_PRESSED { b'@' } else { b'2' }),
157                                0x04 => Some(if SHIFT_PRESSED { b'#' } else { b'3' }),
158                                0x05 => Some(if SHIFT_PRESSED { b'$' } else { b'4' }),
159                                0x06 => Some(if SHIFT_PRESSED { b'%' } else { b'5' }),
160                                0x07 => Some(if SHIFT_PRESSED { b'^' } else { b'6' }),
161                                0x08 => Some(if SHIFT_PRESSED { b'&' } else { b'7' }),
162                                0x09 => Some(if SHIFT_PRESSED { b'*' } else { b'8' }),
163                                0x0a => Some(if SHIFT_PRESSED { b'(' } else { b'9' }),
164                                0x0b => Some(if SHIFT_PRESSED { b')' } else { b'0' }),
165                                0x0c => Some(if SHIFT_PRESSED { b'_' } else { b'-' }),
166                                0x0d => Some(if SHIFT_PRESSED { b'+' } else { b'=' }),
167
168                                // Буквы (если Shift нажат — переводим в заглавные)
169                                0x10 => Some(if SHIFT_PRESSED { b'Q' } else { b'q' }),
170                                0x11 => Some(if SHIFT_PRESSED { b'W' } else { b'w' }),
171                                0x12 => Some(if SHIFT_PRESSED { b'E' } else { b'e' }),
172                                0x13 => Some(if SHIFT_PRESSED { b'R' } else { b'r' }),
173                                0x14 => Some(if SHIFT_PRESSED { b'T' } else { b't' }),
174                                0x15 => Some(if SHIFT_PRESSED { b'Y' } else { b'y' }),
175                                0x16 => Some(if SHIFT_PRESSED { b'U' } else { b'u' }),
176                                0x17 => Some(if SHIFT_PRESSED { b'I' } else { b'i' }),
177                                0x18 => Some(if SHIFT_PRESSED { b'O' } else { b'o' }),
178                                0x19 => Some(if SHIFT_PRESSED { b'P' } else { b'p' }),
179                                0x1e => Some(if SHIFT_PRESSED { b'A' } else { b'a' }),
180                                0x1f => Some(if SHIFT_PRESSED { b'S' } else { b's' }),
181                                0x20 => Some(if SHIFT_PRESSED { b'D' } else { b'd' }),
182                                0x21 => Some(if SHIFT_PRESSED { b'F' } else { b'f' }),
183                                0x22 => Some(if SHIFT_PRESSED { b'G' } else { b'g' }),
184                                0x23 => Some(if SHIFT_PRESSED { b'H' } else { b'h' }),
185                                0x24 => Some(if SHIFT_PRESSED { b'J' } else { b'j' }),
186                                0x25 => Some(if SHIFT_PRESSED { b'K' } else { b'k' }),
187                                0x26 => Some(if SHIFT_PRESSED { b'L' } else { b'l' }),
188                                0x2c => Some(if SHIFT_PRESSED { b'Z' } else { b'z' }),
189                                0x2d => Some(if SHIFT_PRESSED { b'X' } else { b'x' }),
190                                0x2e => Some(if SHIFT_PRESSED { b'C' } else { b'c' }),
191                                0x2f => Some(if SHIFT_PRESSED { b'V' } else { b'v' }),
192                                0x30 => Some(if SHIFT_PRESSED { b'B' } else { b'b' }),
193                                0x31 => Some(if SHIFT_PRESSED { b'N' } else { b'n' }),
194                                0x32 => Some(if SHIFT_PRESSED { b'M' } else { b'm' }),
195
196                                // Пробел
197                                0x39 => Some(b' '),
198                                _ => None,
199                            };
200
201                            if let Some(byte) = key {
202                                if CMD_LEN < CMD_BUFFER_SIZE - 1 {
203                                    CMD_BUFFER[CMD_LEN] = byte;
204                                    CMD_LEN += 1;
205                                    print!("{}", byte as char);
206                                }
207                            }
208                        }
209                    }
210                }
211            }
212            // Все остальные брейк-коды (отпускания клавиш) просто игнорируем
213            _ => {}
214        }
215    }
216
217    // Сообщаем PIC, что обработка завершена
218    unsafe {
219        PICS.lock()
220            .notify_end_of_interrupt(InterruptIndex::Keyboard.as_u8());
